1//! File and JSON helper functions used by the tokenizer API.
2//!
3//! These helpers keep all disk interaction in one module so tokenizer behavior is
4//! easier to test and easier to explain in a tutorial setting.
5
6use crate::tokenizer::{ArtifactPath, Result, TokenizerError};
7use std::fs::read_to_string;
8
9/// Reads a UTF-8 text file and wraps any `io::Error` in
10/// [`TokenizerError::FileRead`].
11///
12/// # Errors
13///
14/// Returns [`TokenizerError::FileRead`] when the file cannot be opened or read.
15pub fn read_text_file(path: &ArtifactPath) -> Result<String> {
16    read_to_string(path).map_err(|source| TokenizerError::FileRead {
17        path: path.clone(),
18        source,
19    })
20}
21
22/// Reads JSON text from `path` and decodes it with `serde_json`.
23///
24/// # Errors
25///
26/// - [`TokenizerError::FileRead`] when the file cannot be read.
27/// - [`TokenizerError::JsonParse`] when parsing fails.
28pub fn read_json_value(path: &ArtifactPath) -> Result<serde_json::Value> {
29    let raw = read_text_file(path)?;
30    serde_json::from_str(&raw).map_err(|source| TokenizerError::JsonParse {
31        path: path.clone(),
32        source,
33    })
34}
35
36/// Serializes and writes JSON to disk using pretty formatting.
37///
38/// # Errors
39///
40/// - [`TokenizerError::FileRead`] when writing fails.
41/// - [`TokenizerError::JsonParse`] when serialization fails.
42pub fn save_json_file(path: &ArtifactPath, value: &serde_json::Value) -> Result<()> {
43    let serialized =
44        serde_json::to_string_pretty(value).map_err(|source| TokenizerError::JsonParse {
45            path: path.clone(),
46            source,
47        })?;
48    std::fs::write(path, serialized).map_err(|source| TokenizerError::FileRead {
49        path: path.clone(),
50        source,
51    })
52}
