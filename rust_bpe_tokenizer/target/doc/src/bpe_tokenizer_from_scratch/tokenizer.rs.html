1//! Public tokenizer API split into small, tutorial-oriented modules.
2//!
3//! This module is the curriculum-facing surface. It introduces explicit domain
4//! types so each API call reads as domain language instead of raw numeric/string
5//! plumbing.
6//!
7//! The high-level flow is:
8//! 1. build or load a vocabulary model,
9//! 2. encode and decode text with predictable special-token handling,
10//! 3. persist or restore learned artifacts.

26/// GPT-2-style marker that prepends a leading space in tokenized symbols.

195/// Validated vocabulary target for the training loop.
196///
197/// The value is intentionally validated at construction time so downstream
198/// functions can assume a minimum-sized symbol table.
199#[derive(Debug, Clone, Copy, PartialEq, Eq)]
200pub struct VocabularySize(usize);
201
202impl VocabularySize {
203    /// Validates and wraps a requested vocabulary size.
204    ///
205    /// # Errors
206    ///
207    /// Returns [`TokenizerError::InvalidVocabularySize`] if `raw` is below
208    /// [`MIN_VOCAB_SIZE`].
209    ///
210    /// # Examples
211    ///
212    /// ```
213    /// use crate::tokenizer::{VocabularySize, MIN_VOCAB_SIZE};
214    /// assert!(VocabularySize::try_new(MIN_VOCAB_SIZE).is_ok());

216    /// ```

223        } else {
224            Ok(Self(raw))
225        }
226    }

229    ///
230    /// # Examples
231    ///
232    /// ```
233    /// use crate::tokenizer::VocabularySize;

235    /// assert_eq!(size.value(), 512);
236    /// ```
237    pub const fn value(self) -> usize {
238        self.0
239    }
240}

547/// Full tokenizer state and conversion tables.
548#[derive(Debug, Default)]
549pub struct BPETokenizerSimple {
550    /// ID → symbol table.
551    pub vocab: BTreeMap<TokenId, TokenSymbol>,
552    /// Symbol → ID table.
553    pub inverse_vocab: HashMap<TokenSymbol, TokenId>,
554    /// Learn-time merges produced during training.
555    pub bpe_merges: HashMap<TokenIdPair, TokenId>,
556    /// Deterministic order used to materialize merged symbols.
557    pub merge_order: Vec<(TokenIdPair, TokenId)>,
558    /// GPT-2 rank table used by OpenAI-compatible merge path.
559    pub bpe_ranks: HashMap<(TokenSymbol, TokenSymbol), MergePriority>,
560}
561
562impl BPETokenizerSimple {
563    /// Creates a tokenizer in a clean, empty state.
564    ///
565    /// # Examples
566    ///
567    /// ```
568    /// use crate::tokenizer::BPETokenizerSimple;
569    /// let tokenizer = BPETokenizerSimple::new();
570    /// assert!(tokenizer.vocab.is_empty());
571    /// ```
572    pub fn new() -> Self {
573        Self::default()
574    }
575
576    /// Train a tokenizer from raw text.
577    ///
578    /// This follows three explicit phases:
579    /// 1. Build an initial symbol table from bytes + observed symbols.
580    /// 2. Add allowed special tokens so the training loop can reserve IDs.
581    /// 3. Learn merge pairs until target vocabulary size is reached.
582    ///
583    /// # Examples
584    ///
585    /// ```
586    /// # use crate::tokenizer::{
587    /// #   BPETokenizerSimple, SpecialTokenSet, Text, VocabularySize, DEFAULT_END_OF_TEXT, TokenSymbol,
588    /// # };
589    /// let mut tokenizer = BPETokenizerSimple::new();
590    /// let text = Text::new("train me");
591    /// let allowed = SpecialTokenSet::new([TokenSymbol::new(DEFAULT_END_OF_TEXT)]);
592    /// tokenizer
593    ///     .train(&text, VocabularySize::try_new(512).unwrap(), &allowed)
594    ///     .unwrap();
595    /// assert!(!tokenizer.vocab.is_empty());
596    /// ```
597    ///
598    /// # Errors
599    ///
600    /// Returns [`TokenizerError::InvalidVocabularySize`] if `vocab_size` is below
601    /// [`MIN_VOCAB_SIZE`], or [`TokenizerError::MissingToken`] if preprocessing
602    /// yields a character with no initial vocabulary mapping.
603    ///
604    /// # Panics
605    ///
606    /// This method does not intentionally panic.
607    pub fn train(
608        &mut self,
609        text: &Text,
610        vocab_size: VocabularySize,
611        allowed_special: &SpecialTokenSet,
612    ) -> Result<()> {
613        self.reset();
614
615        let processed_text = preprocess_spaces_as_leading_marker(text).into_inner();
616        let mut symbols: Vec<TokenSymbol> = (0u32..=255)
617            .filter_map(std::char::from_u32)
618            .map(TokenSymbol::from_char)
619            .collect();
620        for symbol in unique_char_symbols(&processed_text) {
621            let token = TokenSymbol::from_char(symbol);
622            if !symbols.contains(&token) {
623                symbols.push(token);
624            }
625        }
626        if !symbols
627            .iter()
628            .any(|symbol| symbol.as_str() == GPT2_SPACE_MARKER)
629        {
630            symbols.push(TokenSymbol::new(GPT2_SPACE_MARKER));
631        }
632
633        for (raw_id, symbol) in symbols.into_iter().enumerate() {
634            let token_id = TokenId::new(raw_id);
635            self.vocab.insert(token_id, symbol.clone());
636            self.inverse_vocab.insert(symbol, token_id);
637        }
638
639        add_allowed_special_tokens(self, allowed_special);
640
641        let mut token_ids: TokenIdSequence = processed_text
642            .chars()
643            .map(|character| {
644                self.inverse_vocab
645                    .get(&TokenSymbol::from_char(character))
646                    .copied()
647                    .ok_or_else(|| TokenizerError::MissingToken(TokenSymbol::from_char(character)))
648            })
649            .collect::<Result<Vec<_>>>()?
650            .into();
651
652        learn_merge_rules(self, &mut token_ids, vocab_size.value())?;
653        append_merged_symbols(self)?;
654
655        Ok(())
656    }
657
658    /// Loads OpenAI-compatible artifacts (`encoder.json` and `vocab.bpe`).
659    ///
660    /// The loader validates GPT-2 anchor tokens and builds merge priority table
661    /// from the textual merge file.
662    ///
663    /// # Errors
664    ///
665    /// Propagates file or JSON failures via [`TokenizerError::FileRead`] and
666    /// [`TokenizerError::JsonParse`], and anchor validation failures as
667    /// [`TokenizerError::InvalidArtifact`] / [`TokenizerError::UnknownSpecialMissing`].
668    pub fn load_vocab_and_merges_from_openai(
669        &mut self,
670        vocab_path: &ArtifactPath,
671        bpe_merges_path: &ArtifactPath,
672    ) -> Result<()> {
673        self.reset();
674
675        let raw_vocab = read_json_value(vocab_path)?;
676        let loaded_vocab: BTreeMap<String, usize> =
677            serde_json::from_value(raw_vocab).map_err(|source| TokenizerError::JsonParse {
678                path: vocab_path.clone(),
679                source,
680            })?;
681
682        for (symbol_text, id) in loaded_vocab {
683            let token_id = TokenId::new(id);
684            let symbol = TokenSymbol::new(symbol_text);
685            self.vocab.insert(token_id, symbol.clone());
686            self.inverse_vocab.insert(symbol, token_id);
687        }
688
689        validate_required_openai_ids(self)?;
690        ensure_control_symbol_mapping(self)?;
691
692        let merge_text = read_text_file(bpe_merges_path)?;
693        parse_openai_merges(self, &merge_text);
694
695        Ok(())
696    }
697
698    /// Persists the current vocabulary and merge rules to JSON files.
699    ///
700    /// The artifacts are intentionally simple key/value structures so they can be
701    /// versioned and reloaded deterministically for teaching and debugging.
702    ///
703    /// # Errors
704    ///
705    /// Returns [`TokenizerError::JsonParse`] when values cannot be serialized.
706    pub fn save_vocab_and_merges(
707        &self,
708        vocab_path: &ArtifactPath,
709        bpe_merges_path: &ArtifactPath,
710    ) -> Result<()> {
711        let serialized_vocab: BTreeMap<usize, String> = self
712            .vocab
713            .iter()
714            .map(|(id, symbol)| (id.value(), symbol.as_str().to_string()))
715            .collect();
716
717        let serialized_merges: Vec<serde_json::Value> = self
718            .merge_order
719            .iter()
720            .map(|(pair, merged_id)| {
721                serde_json::json!({
722                    "pair": [pair.left().value(), pair.right().value()],
723                    "new_id": merged_id.value(),
724                })
725            })
726            .collect();
727
728        save_json_file(
729            vocab_path,
730            &serde_json::to_value(serialized_vocab).map_err(|source| {
731                TokenizerError::JsonParse {
732                    path: vocab_path.clone(),
733                    source,
734                }
735            })?,
736        )?;
737        save_json_file(
738            bpe_merges_path,
739            &serde_json::to_value(serialized_merges).map_err(|source| {
740                TokenizerError::JsonParse {
741                    path: bpe_merges_path.clone(),
742                    source,
743                }
744            })?,
745        )?;
746
747        Ok(())
748    }
749
750    /// Loads artifacts created by [`Self::save_vocab_and_merges`].
751    ///
752    /// # Errors
753    ///
754    /// Returns:
755    /// - [`TokenizerError::FileRead`] and [`TokenizerError::JsonParse`] for I/O/JSON issues.
756    /// - [`TokenizerError::InvalidArtifact`] when the serialized shape is invalid.
757    pub fn load_vocab_and_merges(
758        &mut self,
759        vocab_path: &ArtifactPath,
760        bpe_merges_path: &ArtifactPath,
761    ) -> Result<()> {
762        self.reset();
763
764        let raw_vocab = read_json_value(vocab_path)?;
765        let loaded_vocab: BTreeMap<String, String> =
766            serde_json::from_value(raw_vocab).map_err(|source| TokenizerError::JsonParse {
767                path: vocab_path.clone(),
768                source,
769            })?;
770
771        for (raw_id, symbol_text) in loaded_vocab {
772            let id: usize = raw_id.parse().map_err(|_| {
773                TokenizerError::InvalidArtifact(format!(
774                    "token id '{raw_id}' in vocab is not numeric"
775                ))
776            })?;
777            let token_id = TokenId::new(id);
778            let symbol = TokenSymbol::new(symbol_text);
779            self.vocab.insert(token_id, symbol.clone());
780            self.inverse_vocab.insert(symbol, token_id);
781        }
782
783        let raw_merges = read_json_value(bpe_merges_path)?;
784        let records: Vec<serde_json::Value> =
785            serde_json::from_value(raw_merges).map_err(|source| TokenizerError::JsonParse {
786                path: bpe_merges_path.clone(),
787                source,
788            })?;
789
790        for record in records {
791            let pair = record
792                .get("pair")
793                .and_then(|value| value.as_array())
794                .ok_or_else(|| {
795                    TokenizerError::InvalidArtifact(
796                        "merge record must include an array field named 'pair'".to_string(),
797                    )
798                })?;
799            if pair.len() != 2 {
800                return Err(TokenizerError::InvalidArtifact(
801                    "merge pair must contain two numeric token IDs".to_string(),
802                ));
803            }
804
805            let Some(left_raw) = pair[0].as_u64() else {
806                return Err(TokenizerError::InvalidArtifact(
807                    "merge pair values must be integer token IDs".to_string(),
808                ));
809            };
810            let Some(right_raw) = pair[1].as_u64() else {
811                return Err(TokenizerError::InvalidArtifact(
812                    "merge pair values must be integer token IDs".to_string(),
813                ));
814            };
815            let Some(new_raw) = record.get("new_id").and_then(|id| id.as_u64()) else {
816                return Err(TokenizerError::InvalidArtifact(
817                    "merge record must include integer field 'new_id'".to_string(),
818                ));
819            };
820
821            let pair = TokenIdPair::new(
822                TokenId::new(left_raw as usize),
823                TokenId::new(right_raw as usize),
824            );
825            let merged_id = TokenId::new(new_raw as usize);
826            self.bpe_merges.insert(pair, merged_id);
827            self.merge_order.push((pair, merged_id));
828        }
829
830        Ok(())
831    }

833    /// Encodes plain text into token IDs.
834    ///
835    /// When `allowed_special` is `Some`, tokens in that set pass through unchanged.
836    /// All other special-token patterns are rejected.
837    ///
838    /// # Errors
839    ///
840    /// Returns [`TokenizerError::MissingToken`] for unmapped characters and
841    /// [`TokenizerError::UnknownSpecialToken`] when disallowed special tokens are
842    /// found in input text.
843    ///
844    /// # Examples
845    ///
846    /// ```
847    /// # use crate::tokenizer::{
848    /// #   BPETokenizerSimple, SpecialTokenSet, Text, VocabularySize, DEFAULT_END_OF_TEXT, TokenSymbol,
849    /// # };
850    /// let mut tokenizer = BPETokenizerSimple::new();
851    /// let text = Text::new("hello world");
852    /// let allowed = SpecialTokenSet::new([TokenSymbol::new(DEFAULT_END_OF_TEXT)]);
853    /// tokenizer
854    ///     .train(&text, VocabularySize::try_new(300).unwrap(), &allowed)
855    ///     .unwrap();
856    /// let encoded = tokenizer.encode(&text, Some(&allowed)).unwrap();
857    /// assert!(!encoded.is_empty());
858    /// ```
859    pub fn encode(
860        &self,
861        text: &Text,
862        allowed_special: Option<&SpecialTokenSet>,
863    ) -> Result<TokenIdSequence> {
864        match allowed_special.filter(|token_set| !token_set.is_empty()) {
865            Some(token_set) => encode_with_special_tokens(self, text, token_set),
866            None => encode_without_special_tokens(self, text),
867        }
868    }
869
870    /// Decodes token IDs back into text.
871    ///
872    /// This mirrors the inverse of the training/encoding rules:
873    /// - control-symbol markers are reconstructed,
874    /// - GPT-2 space markers are converted back to `' '`.
875    ///
876    /// # Errors
877    ///
878    /// Returns [`TokenizerError::MissingTokenId`] when a token id has no
879    /// vocabulary entry.
880    ///
881    /// # Examples
882    ///
883    /// ```
884    /// # use crate::tokenizer::{
885    /// #     BPETokenizerSimple, SpecialTokenSet, Text, VocabularySize, DEFAULT_END_OF_TEXT, TokenSymbol,
886    /// # };
887    /// let mut tokenizer = BPETokenizerSimple::new();
888    /// let input = Text::new("decode");
889    /// let allowed = SpecialTokenSet::new([TokenSymbol::new(DEFAULT_END_OF_TEXT)]);
890    /// tokenizer
891    ///     .train(&input, VocabularySize::try_new(300).unwrap(), &allowed)
892    ///     .unwrap();
893    /// let ids = tokenizer.encode(&input, Some(&allowed)).unwrap();
894    /// let decoded = tokenizer.decode(&ids).unwrap();
895    /// assert_eq!(decoded, input);
896    /// ```
897    pub fn decode(&self, token_ids: &TokenIdSequence) -> Result<Text> {
898        let mut output = String::new();
899        let newline_id = TokenId::new(GPT2_NEWLINE_ID);
900        let carriage_id = TokenId::new(GPT2_CARRIAGE_RETURN_ID);
901
902        for token_id in token_ids.as_slice() {
903            let symbol = self
904                .vocab
905                .get(token_id)
906                .ok_or(TokenizerError::MissingTokenId(*token_id))?;
907            let token = symbol.as_str();
908
909            if token == "\r" || *token_id == carriage_id {
910                output.push('\r');
911            } else if token_id == &newline_id || token == "\n" {
912                if !output.ends_with(' ') {
913                    output.push(' ');
914                }
915                output.push('\n');
916            } else if let Some(rest) = token.strip_prefix(GPT2_SPACE_MARKER) {
917                output.push(' ');
918                output.push_str(rest);
919            } else {
920                output.push_str(token);
921            }
922        }
923
924        Ok(Text::new(output))
925    }

927    /// Returns the ID for a user-facing symbol, if present.
928    ///
929    /// # Examples
930    ///
931    /// ```
932    /// use crate::tokenizer::{BPETokenizerSimple, TokenSymbol};
933    /// let tokenizer = BPETokenizerSimple::new();
934    /// assert_eq!(tokenizer.get_special_token_id(&TokenSymbol::new("<|missing|>")), None);
935    /// ```
936    pub fn get_special_token_id(&self, token: &TokenSymbol) -> Option<TokenId> {
937        self.inverse_vocab.get(token).copied()
938    }

940    /// Finds the most frequent adjacent ID pair (for educational demonstration).
941    ///
942    /// If no adjacent pair exists, returns `None`.
943    ///
944    /// # Examples
945    /// ```
946    /// use crate::tokenizer::{BPETokenizerSimple, TokenId, TokenIdSequence};
947    /// let ids = TokenIdSequence::from_vec(vec![TokenId::new(1), TokenId::new(2), TokenId::new(1)]);
948    /// assert_eq!(
949    ///     BPETokenizerSimple::find_freq_pair(&ids),
950    ///     Some(crate::tokenizer::TokenIdPair::new(TokenId::new(1), TokenId::new(2)))
951    /// );

953    pub fn find_freq_pair(token_ids: &TokenIdSequence) -> Option<TokenIdPair> {

955    }

958        self.vocab.clear();
959        self.inverse_vocab.clear();

966#[derive(Debug, Error)]
967pub enum TokenizerError {
968    /// Returned when `train()` receives a vocabulary target smaller than
969    /// [`MIN_VOCAB_SIZE`].
970    #[error("training vocabulary size {requested} is below minimum {minimum}")]
971    InvalidVocabularySize {
972        requested: VocabularySize,
973        minimum: VocabularySize,
974    },
975
976    /// Returned when a source symbol is absent from the vocabulary map.
977    #[error("token '{0}' is not present in the vocabulary")]
978    MissingToken(TokenSymbol),
979
980    /// Returned when an id does not resolve through the ID→symbol vocabulary table.
981    #[error("token id '{0}' is not present in the vocabulary")]
982    MissingTokenId(TokenId),
983
984    /// Returned when an input contains a special token that is not in the allowed set.
985    #[error("special token '{0}' is not allowed in this input")]
986    UnknownSpecialToken(TokenSymbol),
987
988    /// Returned when a persisted artifact has structurally invalid contents.
989    #[error("invalid artifact file: {0}")]
990    InvalidArtifact(String),
991
992    /// Returned when reading artifact files fails.
993    #[error("could not read artifact '{path}': {source}")]
994    FileRead {
995        path: ArtifactPath,
996        source: std::io::Error,
997    },
998
999    /// Returned when JSON encoding/decoding fails.
1000    #[error("could not parse JSON artifact '{path}': {source}")]
1001    JsonParse {
1002        path: ArtifactPath,
1003        source: serde_json::Error,
1004    },
1005
1006    /// Returned when a known special marker is required by a loader but absent.
1007    #[error("special token '{0}' is missing in this tokenizer")]
1008    UnknownSpecialMissing(TokenSymbol),
1009}
